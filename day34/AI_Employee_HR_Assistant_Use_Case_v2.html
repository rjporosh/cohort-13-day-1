Use Case: AI-Powered Employee HR Assistant

Agentic AI · Function Calling · OpenAI Responses API

1. Business use case

An organization wants to develop an AI-powered Employee HR Assistant that allows employees to access HR information through natural language. Traditionally, employees contact HR, open an HR portal, or search internal systems to check salary, leave, attendance, manager details, holidays, payroll status, job postings, and training opportunities. The proposed assistant simplifies these interactions by understanding employee requests, selecting appropriate Python function(s), retrieving information, and generating a friendly response.

Traditionally, employees need to:

  • Contact the HR department to check salary.
  • Access an HR portal to check leave balance.
  • Contact a manager for department information.
  • Search internal systems for company holidays.
  • Contact HR to check payroll status.
  • Browse internal job postings and training opportunities.

Proposed solution

An employee can ask questions using natural language:

  • What is my salary this month?
  • How many casual leaves do I have?
  • Who is my reporting manager?
  • Show me the available training courses.
  • Tell me about my profile, leave balance, and payroll status.

The AI assistant understands the request, selects the appropriate Python function or functions, retrieves the required information, and generates a user-friendly response.

5. What makes this an Agentic AI application?

This is more than a conventional chatbot. A conventional chatbot might rely on predefined intents and manually written routing logic, such as checking whether the request contains the word “salary” or “leave.” Our AI agent uses the LLM to interpret the request and select the relevant tool. Different phrases such as “What's my pay?”, “How much do I earn monthly?”, “Tell me my compensation,” or “Can I see my salary details?” may all lead to the same tool without separate keyword rules.

Important distinction: This is a tool-using AI agent with a bounded set of capabilities. It is not yet a fully autonomous enterprise agent.

6. Scope and assumptions

  • The 10 Python methods use mock employee data.
  • The LLM selects tools based on natural language.
  • Python executes the business functions.
  • The LLM produces the final formatted response.
  • The agent may execute multiple tools for a single request.
Production consideration: In a real enterprise implementation, the functions would integrate with HR databases or APIs, enforce authentication and authorization, validate tool arguments, and ensure sensitive data is accessible only to authorized employees.

# -------------------------------------------------------------------
# 10 Python tools: deterministic business/data-access functions

def get_employee_info(employee_id: str) -> dict:
    """Return the employee's basic profile."""
    employee = EMPLOYEES.get(employee_id)
    if not employee:
        return {"status": "Employee not found", "employee_id": employee_id}
    return employee


def get_salary_info(employee_id: str) -> dict:
    """Return current salary information for an employee."""
    salary = SALARIES.get(employee_id)
    if not salary:
        return {"status": "Salary information not found", "employee_id": employee_id}
    return {"employee_id": employee_id, **salary}


def get_leave_balance(employee_id: str) -> dict:
    """Return remaining annual, sick, and casual leave balances."""
    balance = LEAVE_BALANCES.get(employee_id)
    if not balance:
        return {"status": "Leave balance not found", "employee_id": employee_id}
    return {"employee_id": employee_id, **balance}


def get_attendance_info(employee_id: str) -> dict:
    """Return attendance information for the current mock reporting month."""
    attendance = ATTENDANCE.get(employee_id)
    if not attendance:
        return {"status": "Attendance information not found", "employee_id": employee_id}
    return {"employee_id": employee_id, **attendance}


def get_department_info(department_id: str) -> dict:
    """Return information about a department."""
    department = DEPARTMENTS.get(department_id)
    if not department:
        return {"status": "Department not found", "department_id": department_id}
    return department


def get_company_holidays() -> dict:
    """Return the list of upcoming company holidays."""
    return {"holidays": COMPANY_HOLIDAYS}


def get_manager_info(employee_id: str) -> dict:
    """Return the employee's reporting manager information."""
    employee = EMPLOYEES.get(employee_id)
    if not employee:
        return {"status": "Employee not found", "employee_id": employee_id}

    manager = MANAGERS.get(employee["manager_id"])
    if not manager:
        return {"status": "Manager information not found", "employee_id": employee_id}
    return {"employee_id": employee_id, "manager": manager}

tool_registry = {
    "get_employee_info": get_employee_info,
    "get_salary_info": get_salary_info,
    "get_leave_balance": get_leave_balance,
    "get_attendance_info": get_attendance_info,

def execute_tool(tool_name: str, arguments: dict) -> dict:
    """Look up and execute a tool using its name and parsed arguments."""
    function = tool_registry.get(tool_name)

    if function is None:
        return {"error": f"Unknown tool: {tool_name}"}

    try:
        return function(**arguments)
    except TypeError as exc:
        # Handles mismatched arguments; production code should log this.
        return {"error": f"Invalid arguments for {tool_name}: {exc}"}


# -------------------------------------------------------------------
# Agent loop: LLM selects tool -> Python executes -> tool result is
# sent back to LLM. The loop continues until the LLM gives a final
# user-facing answer.
# -------------------------------------------------------------------

def run_agent(user_request: str, employee_id: str) -> str:
    instructions = f"""
You are an Employee HR Assistant.

The authenticated employee ID for this session is {employee_id}.
Use this ID for employee-specific tool calls. Do not accept a different
employee ID from the user's message as authorization to access another
employee's records.

Use the available tools when the request requires HR information.
You may call more than one tool if needed.
Do not invent HR data. Use only the tool results.
If the request is outside the tools' capabilities, explain that politely.
After receiving tool results, produce a concise, helpful, user-friendly answer.
"""

    # LLM Call 1: understand request and choose tool(s), or answer directly.
    response = client.responses.create(
        model=MODEL,
        instructions=instructions,
        input=user_request,
        tools=available_tools,
    )

    # Continue until the model returns a final answer with no function calls.
    while True:
        function_calls = [
            item for item in response.output
            if item.type == "function_call"
        ]

        if not function_calls:
            return response.output_text

        tool_outputs = []

        # Pure Python: execute each function call returned by the LLM.
        for call in function_calls:
            arguments = json.loads(call.arguments)

            # For employee-specific tools, enforce session employee ID.
            if "employee_id" in arguments:
                arguments["employee_id"] = employee_id

            result = execute_tool(call.name, arguments)

            tool_outputs.append({
                "type": "function_call_output",
                "call_id": call.call_id,
                "output": json.dumps(result),
            })

        # LLM Call: process tool results, potentially select more tools,
        # or produce the final user-friendly response.
        response = client.responses.create(
            model=MODEL,
            previous_response_id=response.id,
            input=tool_outputs,
            tools=available_tools,
        )


def main() -> None:
    # In production, employee_id must come from authenticated session
    # identity, not from arbitrary user-entered text.
    authenticated_employee_id = "E002"

    user_request = (
        "Tell me about my profile, my manager, and my leave balance."
    )

    final_answer = run_agent(
        user_request=user_request,
        employee_id=authenticated_employee_id,
    )

    print("\nHR Assistant:")
    print(final_answer)


if __name__ == "__main__":
    main()
